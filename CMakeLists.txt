cmake_minimum_required(VERSION 3.16)
project(NewsFeedEngine VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_executable(news_feed src/main.cpp src/storage/FileManager.cpp)
include(CTest)
if(BUILD_TESTING)
  add_executable(news_feed_tests tests/phase9_tests.cpp)
  add_test(NAME regression COMMAND news_feed_tests)
endif()
